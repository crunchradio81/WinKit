cmake_minimum_required(VERSION 3.16)
project(winkit_a1_native LANGUAGES CXX)
enable_testing()
add_library(winkit_disk STATIC ../android/app/src/main/cpp/disk_image.cpp)
target_include_directories(winkit_disk PUBLIC ../android/app/src/main/cpp)
target_compile_features(winkit_disk PUBLIC cxx_std_17)
target_compile_options(winkit_disk PRIVATE -Wall -Wextra -Werror)
add_executable(test_disk test_disk.cpp)
target_link_libraries(test_disk PRIVATE winkit_disk)
add_test(NAME disk_image_safety COMMAND test_disk)
